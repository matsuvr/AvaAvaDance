cmake_minimum_required(VERSION 3.20)
project(AvaDanceNative C CXX)
add_library(AvaFbx SHARED ava_fbx.c ufbx.c)
target_compile_definitions(AvaFbx PRIVATE UFBX_NO_STDIO)
set_property(TARGET AvaFbx PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(AvaMp4 SHARED ava_mp4.cpp)
target_link_libraries(AvaMp4 PRIVATE mfplat mfreadwrite mfuuid ole32)
set_property(TARGET AvaMp4 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set(BULLET_PHYSICS_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/bullet-3.25" CACHE PATH "Bullet 3.25 source checkout")
set(BULLET_VERSION 3.25)
set(BUILD_SHARED_LIBS OFF)
foreach(module IN ITEMS LinearMath BulletCollision BulletDynamics)
    add_subdirectory("${BULLET_PHYSICS_SOURCE_DIR}/src/${module}" "bullet-libs/${module}" EXCLUDE_FROM_ALL)
    set_property(TARGET ${module} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    if(MSVC)
        target_compile_options(${module} PRIVATE /MP)
    endif()
endforeach()
add_library(AvaBullet SHARED ava_bullet.cpp)
target_include_directories(AvaBullet PRIVATE "${BULLET_PHYSICS_SOURCE_DIR}/src")
target_link_libraries(AvaBullet PRIVATE BulletDynamics BulletCollision LinearMath)
set_property(TARGET AvaBullet PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
