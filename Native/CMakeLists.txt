cmake_minimum_required(VERSION 3.20)
project(AvaDanceNative C CXX)
set(EFFEKSEER_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/effekseer-1808" CACHE PATH "Effekseer 1.80.8 source checkout")
foreach(option IN ITEMS BUILD_EXAMPLES BUILD_TOOLS BUILD_GL BUILD_DX9 BUILD_DX12 NETWORK_ENABLED USE_XAUDIO2 USE_DSOUND)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${EFFEKSEER_SOURCE_DIR}" effekseer-libs EXCLUDE_FROM_ALL)
foreach(module IN ITEMS Effekseer EffekseerRendererCommon EffekseerRendererDX11)
    target_compile_options(${module} PRIVATE /MP /utf-8)
endforeach()
add_library(AvaEffekseer SHARED ava_effekseer.cpp)
target_compile_features(AvaEffekseer PRIVATE cxx_std_17)
target_link_libraries(AvaEffekseer PRIVATE EffekseerRendererDX11 d3d11 d3dcompiler dxguid)
set_property(TARGET AvaEffekseer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(AvaFbx SHARED ava_fbx.c ufbx.c)
target_compile_definitions(AvaFbx PRIVATE UFBX_NO_STDIO)
set_property(TARGET AvaFbx PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(AvaRenderBridge SHARED ava_render.cpp ava_animation.cpp)
target_compile_features(AvaRenderBridge PRIVATE cxx_std_17)
find_path(UNITY_PLUGIN_API IUnityGraphics.h HINTS "$ENV{UNITY_PLUGIN_API}"
    "$ENV{ProgramFiles}/Unity/Hub/Editor/2022.3.22f1/Editor/Data/PluginAPI")
if(UNITY_PLUGIN_API)
    target_include_directories(AvaRenderBridge PRIVATE "${UNITY_PLUGIN_API}")
    target_compile_definitions(AvaRenderBridge PRIVATE AVA_UNITY_DEVICE_EVENTS)
endif()
target_link_libraries(AvaRenderBridge PRIVATE d3d11 d3dcompiler dxgi dxguid ole32 gdiplus shlwapi)
set_property(TARGET AvaRenderBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(AvaWindows SHARED ava_windows.cpp)
target_compile_features(AvaWindows PRIVATE cxx_std_17)
target_link_libraries(AvaWindows PRIVATE shell32 user32)
target_compile_options(AvaWindows PRIVATE /utf-8)
set_property(TARGET AvaWindows PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(AvaMp4 SHARED ava_mp4.cpp)
target_link_libraries(AvaMp4 PRIVATE mfplat mfreadwrite mfuuid ole32)
set_property(TARGET AvaMp4 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
